// =============================================================================
//  EyeTrace  --  eye decal tracing helper for ZBrush 2022 / 2025
//  Build with build.py -> dist/EyeTrace.txt (pure ASCII). Install into
//  ZStartup/Macros/EyeTrace/ ; ZBrush compiles it at startup and it shows
//  as  ZPlugin > EyeTrace (Chinese title).
//
//  How it works (all behaviours verified on 2022.0.5 and 2025.0.1):
//   * Old draft stays on the BASE polypaint. "Start" duplicates the SubTool as
//     a hidden backup (<name>ETDraft), then fills the base with the clean
//     colour at (100 - visibility)% so the draft is faded.
//   * A new layer "ETInk-r-g-b" is created on top. Layer polypaint is stored
//     as absolute colour, so the ink does not depend on the faded draft.
//   * "Apply" re-projects the draft from the backup (ProjectAll, other
//     SubTools hidden) and fades again with the new visibility.
//   * "Finish" fills the base with the clean colour at 100% -> only ink left.
//   * FillObject / ProjectAll go to the base only when the selected layer is
//     NOT recording and hidden. IModGet on the layer item: bit0 = REC,
//     bit1 = visible. IClick at (W-24) toggles REC, (W-5) toggles the eye.
//   * Masking is respected by FillObject / ProjectAll (only unmasked area).
//  Numeric variable names contain no underscore (2025 [If] parser issue).
// =============================================================================

// ---- debug log ----
[RoutineDef,ETLog,
	[If,etDbg==1,
		[If,etDbgN==0,[MemDelete,ETDbgMem][MemCreate,ETDbgMem,65536,0][VarSet,etDbgOff,0]]
		[VarInc,etDbgN]
		[VarSet,etLg,[MemWriteString,ETDbgMem,[StrMerge,s,[StrFromAsc,10]],etDbgOff]]
		[VarSet,etDbgOff,etDbgOff+etLg]
		[MemSaveToFile,ETDbgMem,etDbgPath,1]
	]
,s]

// ---- generic helpers ----
[RoutineDef,ETRename,
	[VarSet,etFU,[FileNameResolvePath,"ZSTARTUP_ZPLUGS64/MacroData/ZFileUtils64.dll"]]
	[If,[IExists,btn],
		[FileExecute,[Var,etFU],RenameSetNext,nm]
		[IPress,btn]
	]
,btn,nm]

// current SubTool name without the trailing "."
[RoutineDef,ETSubName,
	[VarSet,etT,[IGetTitle,Tool:Item Info]]
	[VarSet,etTmp,[StrLength,etT]]
	[If,etTmp>0,
		[If,[StrFind,".",etT,etTmp-1]==(etTmp-1),
			[VarSet,etT,[StrExtract,etT,0,etTmp-2]]
		]
	]
]

// make sure a PolyMesh3D is in Edit mode, at the highest SDiv
[RoutineDef,ETCheck,
	[VarSet,etOK,0]
	[If,[IGet,Transform:Edit],
		[If,[IExists,Tool:Geometry:SDiv],
			[If,[IGetMax,Tool:Geometry:SDiv]>1,
				[ISet,Tool:Geometry:SDiv,[IGetMax,Tool:Geometry:SDiv]]
			]
		]
		[If,[IExists,Tool:Layers:New],
			[If,[IsEnabled,Tool:Layers:New],[VarSet,etOK,1]]
		]
	]
	[RoutineCall,ETLog,[StrMerge,"check ok=",etOK," sub=",[SubToolGetActiveIndex]," of ",[SubToolGetCount]]]
]

// is the selected layer the ink layer? -> etOK, etLn, etLp
[RoutineDef,ETInkSelected,
	[VarSet,etOK,0]
	[VarSet,etLn,""]
	[If,[IExists,Tool:Layers:Layer Intensity],
		[If,[IsEnabled,Tool:Layers:Layer Intensity],
			[VarSet,etLn,[IGetTitle,Tool:Layers:Layer Intensity]]
			[VarSet,etLp,[IGetTitle,Tool:Layers:Layer Intensity,1]]
			[If,[StrFind,"ETInk",etLn,0]==0,[VarSet,etOK,1]]
		]
	]
]

// the ink layer must be the selected layer (walking the list with SelectUp/Down is unreliable)
[RoutineDef,ETFindInk,
	[RoutineCall,ETInkSelected]
	[RoutineCall,ETLog,[StrMerge,"findink ok=",etOK," name=",etLn]]
]

// parse clean colour from "ETInk-r-g-b"
[RoutineDef,ETParseInk,
	[RoutineCall,ETLog,[StrMerge,"parse in ln=",etLn]]
	[VarSet,etQa,[StrFind,"-",etLn,0]]
	[If,etQa>0,
		[VarSet,etQb,[StrFind,"-",etLn,etQa+1]]
		[VarSet,etQc,[StrFind,"-",etLn,etQb+1]]
		[If,(etQb>etQa)*(etQc>etQb),
			// NOTE: never nest [StrLength,..]-1 inside an argument on 2025 (silently stops)
			[VarSet,etLen,[StrLength,etLn]]
			[VarSet,etCR,[Val,[StrExtract,etLn,etQa+1,etQb-1]]]
			[VarSet,etCG,[Val,[StrExtract,etLn,etQb+1,etQc-1]]]
			[VarSet,etCB,[Val,[StrExtract,etLn,etQc+1,etLen-1]]]
		]
	]
	[RoutineCall,ETLog,[StrMerge,"clean=",etCR,",",etCG,",",etCB]]
]

// read current main colour into etUR/etUG/etUB (etOK=0 if impossible)
[RoutineDef,ETReadColor,
	[VarSet,etOK,0]
	[If,[IExists,Color:R],
		[VarSet,etUR,[IGet,Color:R]]
		[VarSet,etUG,[IGet,Color:G]]
		[VarSet,etUB,[IGet,Color:B]]
		[VarSet,etOK,1]
	]
	[RoutineCall,ETLog,[StrMerge,"color ok=",etOK," ",etUR,",",etUG,",",etUB]]
]

// locate backup SubTool <name>ETDraft -> etBk (-1 if none); keeps etA active
[RoutineDef,ETFindBackup,
	[VarSet,etBk,-1]
	[VarSet,etA,[SubToolGetActiveIndex]]
	[RoutineCall,ETSubName]
	[VarSet,etName,etT]
	[VarSet,etBkName,[StrMerge,etName,"ETDraft"]]
	[VarSet,etN,[SubToolGetCount]]
	[VarSet,etI,0]
	[Loop,etN,
		[If,etI!=etA,
			[SubToolSelect,etI]
			[RoutineCall,ETSubName]
			[If,[StrLength,etT]==[StrLength,etBkName],
				[If,[StrFind,etBkName,etT,0]==0,[VarSet,etBk,etI][LoopExit]]
			]
		]
		[VarInc,etI]
	]
	[SubToolSelect,etA]
	[RoutineCall,ETLog,[StrMerge,"backup ",etBkName," idx=",etBk," active=",etA]]
]

// before touching the base: stop recording + hide the selected layer
[RoutineDef,ETBaseBegin,
	[VarSet,etHasL,0]
	[If,[IExists,Tool:Layers:Layer Intensity],
		[If,[IsEnabled,Tool:Layers:Layer Intensity],[VarSet,etHasL,1]]
	]
	// layer item IModGet (measured): 1 = recording (visible), 2 = visible only, 0 = hidden
	// IClick at W-24 toggles REC (measured), W-5 toggles the eye
	[If,etHasL,
		[VarSet,etLp,[IGetTitle,Tool:Layers:Layer Intensity,1]]
		[VarSet,etLw,[IWidth,etLp]]
		[VarSet,etLh,[IHeight,etLp]]
		[VarSet,etModz,[IModGet,etLp]]
		[VarSet,etMd,etModz]
		[If,etMd==1,[IClick,etLp,etLw-24,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		[If,etMd==2,[IClick,etLp,etLw-5,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		[RoutineCall,ETLog,[StrMerge,"basebegin ",etLp," mod0=",etModz," now=",etMd]]
	]
]

// afterwards: show the layer again and resume recording if it was on
[RoutineDef,ETBaseEnd,
	[If,etHasL,
		[VarSet,etMd,[IModGet,etLp]]
		[If,etModz>0,
			[If,etMd==0,[IClick,etLp,etLw-5,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		]
		[If,etModz==1,
			[If,etMd==2,[IClick,etLp,etLw-24,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		]
		[RoutineCall,ETLog,[StrMerge,"baseend mod=",etMd]]
	]
]

// make the selected (ink) layer visible AND recording, so new strokes go into it
[RoutineDef,ETRecOn,
	[If,[IsEnabled,Tool:Layers:Layer Intensity],
		[VarSet,etLp,[IGetTitle,Tool:Layers:Layer Intensity,1]]
		[VarSet,etLw,[IWidth,etLp]]
		[VarSet,etLh,[IHeight,etLp]]
		[VarSet,etMd,[IModGet,etLp]]
		[RoutineCall,ETLog,[StrMerge,"recon start mod=",etMd]]
		[If,etMd==0,[IClick,etLp,etLw-5,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		[RoutineCall,ETLog,[StrMerge,"recon after eye mod=",etMd]]
		[If,etMd==2,[IClick,etLp,etLw-24,etLh/2][VarSet,etMd,[IModGet,etLp]]]
		[RoutineCall,ETLog,[StrMerge,"recon mod=",etMd]]
	]
]

// Set the main colour. Besides IColorSet, also push the Color:R/G/B sliders,
// otherwise sliders the user moved by hand can put their old value back
// (user report on 2025: picked black, painted white).
[RoutineDef,ETSetColor,
	[IColorSet,cr,cg,cb]
	[If,[IExists,Color:R],
		[ISet,Color:R,cr]
		[ISet,Color:G,cg]
		[ISet,Color:B,cb]
	]
	[IColorSet,cr,cg,cb]
,cr,cg,cb]

// FillObject with colour (r,g,b) at Rgb Intensity pct (colour only, no material)
[RoutineDef,ETFill,
	[If,pct>0,
		[VarSet,etRI,[IGet,Draw:Rgb Intensity]]
		[VarSet,etMrgb,[IGet,Draw:Mrgb]]
		[IPress,Draw:Rgb]
		[RoutineCall,ETSetColor,fr,fg,fb]
		[ISet,Draw:Rgb Intensity,pct]
		[IPress,Color:FillObject]
		[ISet,Draw:Rgb Intensity,etRI]
		[If,etMrgb,[IPress,Draw:Mrgb]]
		[RoutineCall,ETLog,[StrMerge,"fill ",fr,",",fg,",",fb," @",pct]]
	]
,fr,fg,fb,pct]

// re-project the draft from backup etBk onto active etA (others hidden)
[RoutineDef,ETProjectDraft,
	[VarSet,etN,[SubToolGetCount]]
	[VarSet,etI,0]
	[Loop,etN,
		[VarSet,etTmp,[SubToolGetStatus,etI]]
		[VarSet,etSt(etI),etTmp]
		[If,((etI==etA)+(etI==etBk))>0,
			[If,(etTmp&1)==0,[VarSet,etTmp,etTmp+1]]
		,
			[If,(etTmp&1)==1,[VarSet,etTmp,etTmp-1]]
		]
		[SubToolSetStatus,etI,etTmp]
		[VarInc,etI]
	]
	[SubToolSelect,etA]
	[VarSet,etPB,[IGet,Tool:SubTool:PA Blur]]
	[ISet,Tool:SubTool:PA Blur,0]
	[IPress,Tool:SubTool:ProjectAll]
	[ISet,Tool:SubTool:PA Blur,etPB]
	[VarSet,etI,0]
	[Loop,etN,
		[SubToolSetStatus,etI,etSt(etI)]
		[VarInc,etI]
	]
	[SubToolSelect,etA]
	[RoutineCall,ETLog,"projected draft"]
]

// Put the brush into tracing state. Zadd/Zsub are always switched off (they
// would change the shape). Rgb Intensity: the user chooses it with the
// "ink strength" slider (default 100). If the brush differs from the slider:
//   ask=1 -> ask the user (set to slider value / keep current), never force
//   ask=0 -> set it to the slider value silently ("Reset brush" button)
// etFix gets a note about what was corrected (Zadd).
[RoutineDef,ETBrushFix,
	[VarSet,etTg,[IGet,#etSlInk]]
	[VarSet,etRIa,[IGet,Draw:Rgb Intensity]]
	[VarSet,etZa,[IGet,Draw:Zadd]]
	[IUnPress,Draw:Zadd]
	[IUnPress,Draw:Zsub]
	[IPress,Draw:Rgb]
	[VarSet,etFix,""]
	[If,etZa,[VarSet,etFix,#etMZadd]]
	[If,etRIa!=etTg,
		[If,ask,
			[NoteIButton,[StrMerge,#etBSetA,etTg]]
			[NoteIButton,[StrMerge,#etBKeepA,etRIa]]
			[VarSet,etRes,[Note,[StrMerge,#etMRgbA,etRIa,#etMRgbB,etTg,#etMRgbC]]]
			[If,etRes==1,[ISet,Draw:Rgb Intensity,etTg]]
		,
			[ISet,Draw:Rgb Intensity,etTg]
		]
	]
	[VarSet,etRIb,[IGet,Draw:Rgb Intensity]]
	[RoutineCall,ETLog,[StrMerge,"brushfix rgbint ",etRIa," -> ",etRIb," target=",etTg," zadd=",etZa]]
,ask]

// "Reset brush" button
[RoutineDef,ETBrushReset,
	[VarSet,etDbgN,0]
	[RoutineCall,ETBrushFix,0]
	[RoutineCall,ETSetColor,0,0,0]
	[RoutineCall,ETInkSelected]
	[If,etOK,[RoutineCall,ETRecOn]]
	[Note,[StrMerge,#etMBrushOK,#etFix]]
]

// ---- actions ----
[RoutineDef,ETStart,
	[VarSet,etDbgN,0]
	[RoutineCall,ETCheck]
	[If,etOK==0,[Note,#etMNoMesh][Exit]]
	[RoutineCall,ETInkSelected]
	[If,etOK==1,[Note,#etMBusy][Exit]]
	[RoutineCall,ETFindBackup]
	[If,etBk>=0,[Note,#etMHasBak][Exit]]
	[RoutineCall,ETReadColor]
	[If,etOK==0,
		[VarSet,etUR,255][VarSet,etUG,255][VarSet,etUB,255]
		[Note,#etMNoColor]
	]
	[VarSet,etCR,etUR][VarSet,etCG,etUG][VarSet,etCB,etUB]
	[VarSet,etV,[IGet,#etSl]]
	[NoteIButton,#etBOK]
	[NoteIButton,#etBCancel]
	[VarSet,etRes,[Note,[StrMerge,#etMStart1,etCR,", ",etCG,", ",etCB,#etMStart2,etV,"%"]]]
	[If,etRes!=1,[Exit]]
	// 1) hidden backup copy
	[VarSet,etA,[SubToolGetActiveIndex]]
	[VarSet,etN,[SubToolGetCount]]
	[IPress,Tool:SubTool:Duplicate]
	[If,[SubToolGetCount]!=(etN+1),[Note,#etMDupFail][Exit]]
	[VarSet,etBk,[SubToolGetActiveIndex]]
	[RoutineCall,ETRename,"Tool:SubTool:Rename",[StrMerge,etName,"ETDraft"]]
	[If,etBk<=etA,[VarSet,etA,etA+1]]
	[SubToolSelect,etA]
	[VarSet,etTmp,[SubToolGetStatus,etBk]]
	[If,(etTmp&1)==1,[SubToolSetStatus,etBk,etTmp-1]]
	[RoutineCall,ETLog,[StrMerge,"dup bk=",etBk," a=",etA]]
	// 2) fade the draft on the base
	[RoutineCall,ETBaseBegin]
	[RoutineCall,ETFill,etCR,etCG,etCB,100-etV]
	[RoutineCall,ETBaseEnd]
	// 3) ink layer
	[IPress,Tool:Layers:New]
	[RoutineCall,ETRename,"Tool:Layers:Rename",[StrMerge,"ETInk-",etCR,"-",etCG,"-",etCB]]
	[RoutineCall,ETBrushFix,1]
	[RoutineCall,ETSetColor,0,0,0]
	[RoutineCall,ETInkSelected]
	[RoutineCall,ETLog,[StrMerge,"start done ink=",etLn," mod=",[IModGet,etLp]]]
	[Note,[StrMerge,#etMStartOK,#etFix]]
]

[RoutineDef,ETApply,
	[VarSet,etDbgN,0]
	[RoutineCall,ETCheck]
	[If,etOK==0,[Note,#etMNoMesh][Exit]]
	[RoutineCall,ETFindBackup]
	[If,etBk<0,[Note,#etMNoBak][Exit]]
	[RoutineCall,ETFindInk]
	[RoutineCall,ETLog,[StrMerge,"after findink etOK=",etOK]]
	[If,etOK==0,[Note,#etMNoInk][Exit]]
	[RoutineCall,ETLog,"apply before parse"]
	[RoutineCall,ETParseInk]
	[RoutineCall,ETReadColor]
	[VarSet,etV,[IGet,#etSl]]
	[RoutineCall,ETBaseBegin]
	[RoutineCall,ETProjectDraft]
	[RoutineCall,ETFill,etCR,etCG,etCB,100-etV]
	[RoutineCall,ETBaseEnd]
	[RoutineCall,ETRecOn]
	[If,etOK,[RoutineCall,ETSetColor,etUR,etUG,etUB]]
	// the fill above restored the user's own Rgb Intensity: check it now
	[RoutineCall,ETBrushFix,1]
	[VarSet,etLen,[StrLength,etFix]]
	[If,etLen>0,
		[Note,[StrMerge,#etMApplyOK," ",etV,"%",#etFix]]
	,
		[NoteBar,[StrMerge,#etMApplyOK," ",etV,"%"]]
	]
]

[RoutineDef,ETFinish,
	[VarSet,etDbgN,0]
	[RoutineCall,ETCheck]
	[If,etOK==0,[Note,#etMNoMesh][Exit]]
	[RoutineCall,ETFindBackup]
	[RoutineCall,ETFindInk]
	[If,etOK==0,[Note,#etMNoInk][Exit]]
	[RoutineCall,ETParseInk]
	[RoutineCall,ETReadColor]
	[RoutineCall,ETBaseBegin]
	[RoutineCall,ETFill,etCR,etCG,etCB,100]
	[RoutineCall,ETBaseEnd]
	[RoutineCall,ETRecOn]
	[If,etOK,[RoutineCall,ETSetColor,etUR,etUG,etUB]]
	[If,etBk>=0,
		[NoteIButton,#etBDel]
		[NoteIButton,#etBKeep]
		[VarSet,etRes,[Note,#etMFinish1]]
		[If,etRes==1,
			[SubToolSelect,etBk]
			[IKeyPress,'2',[IPress,Tool:SubTool:Delete]]
			[If,etBk<etA,[VarSet,etA,etA-1]]
			[SubToolSelect,etA]
			[RoutineCall,ETLog,"backup deleted"]
		]
	,
		[Note,#etMFinishOK]
	]
]

[RoutineDef,ETRestore,
	[VarSet,etDbgN,0]
	[RoutineCall,ETCheck]
	[If,etOK==0,[Note,#etMNoMesh][Exit]]
	[RoutineCall,ETFindBackup]
	[If,etBk<0,[Note,#etMNoBak][Exit]]
	[RoutineCall,ETFindInk]
	[RoutineCall,ETBaseBegin]
	[RoutineCall,ETProjectDraft]
	[RoutineCall,ETBaseEnd]
	[RoutineCall,ETInkSelected]
	[If,etOK,[RoutineCall,ETRecOn]]
	[NoteBar,#etMRestoreOK]
]
